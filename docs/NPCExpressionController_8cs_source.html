139 public void SetExpression(ExpressionType expression)
140 {
141 if (isDead && expression != ExpressionType.Death) return;
142 if (expression == activeExpression || isApplying) return;
143 if (isGuard && expression == ExpressionType.Scared) return;
144 if (!isGuard && expression == ExpressionType.Angry) return;
145
146 isApplying = true;
147 ResolveParents();
148 CaptureOriginals(false);
149
150 if (debugLog) Debug.Log("[NPCExpression] Request -> " + expression + " (isGuard=" + isGuard + ")", this);
151
152 if (expression == ExpressionType.Scared)
153 {
154 ApplyExpression(SafeEyes(featurePack != null ? featurePack.scaredEyes : null),
155 SafeMouths(featurePack != null ? featurePack.scaredMouths : null));
156 }
157 else if (expression == ExpressionType.Angry)
158 {
159 ApplyExpression(SafeEyes(featurePack != null ? featurePack.angryEyes : null),
160 SafeMouths(featurePack != null ? featurePack.angryMouths : null));
161 }
162 else if (expression == ExpressionType.Death)
163 {
164 isDead = true;
165 ApplyExpression(SafeEyes(featurePack != null ? featurePack.deathEyes : null),
166 SafeMouths(featurePack != null ? featurePack.deathMouths : null));
167 }
168 else
169 {
170 RevertToOriginals();
171 }
172
173 activeExpression = expression;
174 isApplying = false;
175 ExpressionChanged?.Invoke(activeExpression);
176 }

214 private void CaptureOriginals(bool force)
215 {
216 if (force)
217 {
218 originalEyeObjects.Clear();
219 originalMouthObjects.Clear();
220 originalsHidden = false;
221 leftEyeT = rightEyeT = mouthT = null;
222 }
223
224 if (originalEyeObjects.Count == 0)
225 CollectByAny(eyesParent != null ? eyesParent : transform, new string[] { "eye", "eyes" }, originalEyeObjects);
226
227 if (originalEyeObjects.Count >= 1)
228 {
229 originalEyeObjects.Sort((GameObject a, GameObject b) => a.transform.localPosition.x.CompareTo(b.transform.localPosition.x));
230
231 leftEyeT = originalEyeObjects[0].transform;
232 leftPos = leftEyeT.localPosition;
233 leftRot = leftEyeT.localRotation;
234 eyeChildScale = leftEyeT.localScale;
235 eyesOrigParent = leftEyeT.parent;
236
237 if (originalEyeObjects.Count >= 2)
238 {
239 rightEyeT = originalEyeObjects[1].transform;
240 rightPos = rightEyeT.localPosition;
241 rightRot = rightEyeT.localRotation;
242 pairCenterPos = (leftPos + rightPos) * 0.5f;
243 }
244 else
245 {
246 rightEyeT = null;
247 pairCenterPos = leftPos;
248 }
249 }
250
251 if (originalMouthObjects.Count == 0)
252 CollectByAny(mouthParent != null ? mouthParent : transform, new string[] { "mouth", "face" }, originalMouthObjects);
253
254 if (originalMouthObjects.Count > 0)
255 {
256 mouthT = originalMouthObjects[0].transform;
257 mouthPos = mouthT.localPosition;
258 mouthRot = mouthT.localRotation;
259 mouthScale = mouthT.localScale;
260 mouthOrigParent = mouthT.parent;
261 }

265 private void ApplyExpression(GameObject[] eyesOptions, GameObject[] mouthOptions)
266 {
267 HideOriginals();
268 ClearExpression();
269
270 GameObject nextEyes = PickOne(eyesOptions);
271 GameObject nextMouth = PickOne(mouthOptions);
272
273 if (nextEyes != null && eyesParent != null && leftEyeT != null)
274 {
275 bool paired = nextEyes.name.ToLower().Contains("eyes");
276 if (paired)
277 {
278 exprEyesPaired = Object.Instantiate(nextEyes, eyesParent);
279 exprEyesPaired.name = ExprPrefix + nextEyes.name;
280
281 Vector3 off = rotateOffsetsWithOriginals ? (leftRot * eyesOffset) : eyesOffset;
282 exprEyesPaired.transform.localPosition = pairCenterPos + off;
283 exprEyesPaired.transform.localRotation = useOriginalRotation ? leftRot : Quaternion.identity;
284 exprEyesPaired.transform.localScale = eyeChildScale;
285 }
286 else
287 {
288 Vector3 offL = rotateOffsetsWithOriginals ? (leftRot * eyesOffset) : eyesOffset;
289
290 GameObject l = Object.Instantiate(nextEyes, eyesParent);
291 l.name = ExprPrefix + nextEyes.name + "_L";
292 l.transform.localPosition = leftPos + offL;
293 l.transform.localRotation = useOriginalRotation ? leftRot : Quaternion.identity;
294 l.transform.localScale = eyeChildScale;
295 exprEyeInstances.Add(l);
296
297 if (rightEyeT != null)
298 {
299 Vector3 offR = rotateOffsetsWithOriginals ? (rightRot * eyesOffset) : eyesOffset;
300 if (mirrorRightEye) offR.x = -offR.x;
301
302 GameObject r = Object.Instantiate(nextEyes, eyesParent);
303 r.name = ExprPrefix + nextEyes.name + "_R";
304 r.transform.localPosition = rightPos + offR;
305 r.transform.localRotation = useOriginalRotation ? rightRot : Quaternion.identity;
306 r.transform.localScale = mirrorRightEye
307 ? new Vector3(-Mathf.Abs(eyeChildScale.x), eyeChildScale.y, eyeChildScale.z)
308 : eyeChildScale;
309 exprEyeInstances.Add(r);
310 }
311 }
312 }
313
314 if (nextMouth != null && mouthParent != null && mouthT != null)
315 {
316 exprMouth = Object.Instantiate(nextMouth, mouthParent);
317 exprMouth.name = ExprPrefix + nextMouth.name;
318
319 Vector3 moff = rotateOffsetsWithOriginals ? (mouthRot * mouthOffset) : mouthOffset;
320 exprMouth.transform.localPosition = mouthPos + moff;
321 exprMouth.transform.localRotation = useOriginalRotation ? mouthRot : Quaternion.identity;
322 exprMouth.transform.localScale = mouthScale;
323 }

500 private void TryHook(object target, bool sub, string[] candidateNames, System.Action<GameObject> cb)
501 {
502 BindingFlags flags = BindingFlags.Instance | BindingFlags.Public | BindingFlags.NonPublic;
503 UnityAction<GameObject> ua = GetUnityAction(cb);
504
505 foreach (string name in candidateNames)
506 {
507 FieldInfo f = target.GetType().GetField(name, flags);
508 if (f != null && f.FieldType.IsGenericType &&
509 f.FieldType.GetGenericTypeDefinition() == typeof(UnityEvent<>))
510 {
511 UnityEvent<GameObject> evField = f.GetValue(target) as UnityEvent<GameObject>;
512 if (evField != null)
513 {
514 if (sub) evField.AddListener(ua); else evField.RemoveListener(ua);
515 }
516 continue;
517 }
518
519 PropertyInfo p = target.GetType().GetProperty(name, flags);
520 if (p != null && p.PropertyType.IsGenericType &&
521 p.PropertyType.GetGenericTypeDefinition() == typeof(UnityEvent<>))
522 {
523 UnityEvent<GameObject> evProp = p.GetValue(target, null) as UnityEvent<GameObject>;
524 if (evProp != null)
525 {
526 if (sub) evProp.AddListener(ua); else evProp.RemoveListener(ua);
527 }
528 }
529 }
