186 private void FillScene()
187 {
188 // Disables all NavMeshes that aren't the main one for spawning.
189 SetNavMeshStates(false, genericNavMesh);
190
191 // Backup check count to protect infinite while loop
192 const int MAX_CHECK_COUNT = MAX_NPC_COUNT + 100;
193 int checkCount = 0;
194
195 Bounds navMeshBounds = genericNavMesh.navMeshData.sourceBounds;
196
197 while (Contract.Instance.Npcs.Count < MAX_NPC_COUNT && checkCount < MAX_CHECK_COUNT)
198 {
199 Vector3 randomPosition = new Vector3(
200 Random.Range(navMeshBounds.min.x, navMeshBounds.max.x),
201 Random.Range(navMeshBounds.min.y, navMeshBounds.max.y),
202 Random.Range(navMeshBounds.min.z, navMeshBounds.max.z)
203 );
204
205 NavMeshHit hit;
206
207 if(NavMesh.SamplePosition(randomPosition, out hit, Mathf.Infinity, 1))
208 {
209 // Spawns an NPC with a random home, at the desired position with a random goal.
210 SpawnRandomNPC(default, hit.position);
211 }
212 else
213 {
214 SpawnRandomNPC();
215 }
216
217 checkCount++;
218 }
219
220 if (checkCount == MAX_CHECK_COUNT)
221 {
222 Debug.LogWarning("Something in NPCSpawner.cs FillScene() is not spawning properly. The while loop is exciting early.");
223 }
224
225 // Enables all NavMeshes when complete.
226 SetNavMeshStates(true);
227 }
