128 private void SpawnNPC(Vector3 home, Vector3 spawnPoint, Vector3 goal)

146 }
147 else if (roll == spawnableTypes.IndexOf(NPCType.GuardLeader))

149 GuardLeader guard = activeNPC.AddComponent<GuardLeader>();
150 guard.SetHomeSpawnGoal(home, spawnPoint, goal);
151 guard.SpawnFollowers(npc, parent, characterCreator);
152 guard.player = player;
153 activeNPC.name = "NPC - Guard";
154 }
155 else if (roll == spawnableTypes.IndexOf(NPCType.Crowd))
156 {
157 Crowd crowd = activeNPC.AddComponent<Crowd>();
158 crowd.SetHomeSpawnGoal(home, spawnPoint, goal);
159 crowd.FindCrowd(crowdPoints);
160 activeNPC.gameObject.name = "NPC - Crowd";
161 }
162
163 characterCreator.SpawnNPCModel(activeNPC.transform, spawnableTypes[roll]);
164 activeNPC.transform.LookAt(parent);

166 Contract.Instance.AddNPC(activeNPC);

195 private void FillScene()
196 {
197 // Disables all NavMeshes that aren't the main one for spawning.
198 SetNavMeshStates(false, genericNavMesh);
199
200 // Backup check count to protect infinite while loop
201 const int MAX_CHECK_COUNT = MAX_NPC_COUNT + 100;
202 int checkCount = 0;
203
204 Bounds navMeshBounds = genericNavMesh.navMeshData.sourceBounds;
205
206 while (Contract.Instance.Npcs.Count < MAX_NPC_COUNT && checkCount < MAX_CHECK_COUNT)
207 {
208 Vector3 randomPosition = new Vector3(
209 Random.Range(navMeshBounds.min.x, navMeshBounds.max.x),
210 Random.Range(navMeshBounds.min.y, navMeshBounds.max.y),
211 Random.Range(navMeshBounds.min.z, navMeshBounds.max.z)
212 );
213
214 NavMeshHit hit;
215
216 if (NavMesh.SamplePosition(randomPosition, out hit, Mathf.Infinity, 1))
217 {
218 // Spawns an NPC with a random home, at the desired position with a random goal.
219 SpawnRandomNPC(default, hit.position);
220 }
221 else
222 {
223 SpawnRandomNPC();
224 }
225
226 checkCount++;
227 }
228
229 if (checkCount == MAX_CHECK_COUNT)
230 {
231 Debug.LogWarning("Something in NPCSpawner.cs FillScene() is not spawning properly. The while loop is exciting early.");
232 }
233
234 // Enables all NavMeshes when complete.
235 SetNavMeshStates(true);
236 }
