144 protected void CheckDistance()
145 {
146 if (agent.hasPath) //waits for generation
147 {
148 distance = 0.0f;
149 Vector3[] corners = agent.path.corners;
150 for (int i = 0; i < corners.Length - 1; i++)
151 {
152 distance += Mathf.Abs((corners[i] - corners[i + 1]).magnitude);
153 }
154
155 if (distance <= endSize)
