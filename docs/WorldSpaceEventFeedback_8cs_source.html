175 IEnumerator AnimateSphereAlphaCoroutine(float targetAlpha)
176 {
177 float startAlpha = playerSphere.material.GetFloat("_Alpha");
178 float alpha = startAlpha;
179 float animLinear = 0;
180 float animCurved;
181
182 while (alpha != targetAlpha)
183 {
184 animLinear += Time.unscaledDeltaTime / sphereAlphaAnimationDuration;
185 animLinear = Mathf.Clamp(animLinear, 0, 1);
186 animCurved = sphereAlphaAnimationCurve.Evaluate(animLinear);
187 alpha = Mathf.Lerp(startAlpha, targetAlpha, animCurved);
188 playerSphere.material.SetFloat("_Alpha", alpha);
189 yield return null;
190 }
191 }

207 animCurved = sphereRadiusAnimationCurve.Evaluate(animLinear);
208 radius = Mathf.Lerp(startRadius, targetRadius, animCurved);
209 playerSphere.transform.localScale = Vector3.one * radius;
