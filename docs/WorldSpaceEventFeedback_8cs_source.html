172 IEnumerator AnimateSphereAlphaCoroutine(float targetAlpha)
173 {
174 float startAlpha = playerSphere.material.GetFloat("_Alpha");
175 float alpha = startAlpha;
176 float animLinear = 0;
177 float animCurved;
178
179 while (alpha != targetAlpha)
180 {
181 animLinear += Time.unscaledDeltaTime / sphereAlphaAnimationDuration;
182 animLinear = Mathf.Clamp(animLinear, 0, 1);
183 animCurved = sphereAlphaAnimationCurve.Evaluate(animLinear);
184 alpha = Mathf.Lerp(startAlpha, targetAlpha, animCurved);
185 playerSphere.material.SetFloat("_Alpha", alpha);
186 yield return null;
187 }
188 }

193 IEnumerator AnimateSphereRadiusCoroutine(float targetRadius)
194 {

203 animLinear = Mathf.Clamp(animLinear, 0, 1);
204 animCurved = sphereRadiusAnimationCurve.Evaluate(animLinear);
205 radius = Mathf.Lerp(startRadius, targetRadius, animCurved);
206 playerSphere.transform.localScale = Vector3.one * radius;
207 yield return null;
