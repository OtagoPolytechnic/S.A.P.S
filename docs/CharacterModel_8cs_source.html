253 public void SetPositionFromPlacement()
254 {
255 if (model == null)
256 {
257 Debug.LogWarning($"Tried setting position of {featureObject.name} but couldn't find the body.");
258 return;
259 }
260
261 if (featureObject == null)
262 {
263 return;
264 }
265
266 if (placement.fixedPosition)
267 {
268 featureObject.transform.localPosition = Vector3.zero;
269 return;
270 }
271
272 BodyMargins margins = model.bodyMargins;
273
274 // get the midpoint of the body, but not above or below the cylindrical area
275 // these two magic numbers are fractionally where capsule caps begin; i.e. where cylinder becomes a semisphere
276 Vector3 originInCylinder = new() { y = Mathf.Clamp(placement.height * margins.height, margins.cylinderBottom, margins.cylinderTop) };
277
278 // account for rounding of the top and bottom of capsule
279 Vector2 distanceFromOrigin = new()
280 {
281 y = Mathf.Abs(placement.height * margins.height - originInCylinder.y)
282 };
283 float radiusScale = 1;
284 if (placement.protruding && distanceFromOrigin.y > 0)
285 {
286 // scales horizontal coords to stick feature to body when height is above or below the capsule crossover point
287 // using trig: hypotenuse = margins.radius, side B = originDistance, looking for length of side A
288 distanceFromOrigin.x = Mathf.Sqrt(Mathf.Pow(margins.radius, 2) - Mathf.Pow(distanceFromOrigin.y, 2));
289 distanceFromOrigin.x = Mathf.Clamp(Mathf.Abs(distanceFromOrigin.x), 0, margins.radius);
290 // accounts for the radius of the horizontal cross section getting smaller as you approach the ends of the capsule
291 radiusScale = Mathf.InverseLerp(0, margins.radius, distanceFromOrigin.x);
292 }
293
294 // roughly get the point on the surface of the body
295 featureObject.transform.localPosition = new(
296 margins.radius * radiusScale * Mathf.Sin(placement.angle),
297 margins.height * placement.height,
298 margins.radius * radiusScale * Mathf.Cos(placement.angle)
299 );
300
301 if (placement.protruding)
302 {
303 // stand up along the angle from origin
304 Vector3 positionOnMesh = featureObject.transform.localPosition;
305 positionOnMesh.y = placement.height * margins.height;
306 featureObject.transform.up = (positionOnMesh - originInCylinder).normalized;
307 }
308 else
309 {
310 // face horizontally away from the midpoint (0, 0)
311 Vector3 direction = new Vector3()
312 {
313 x = featureObject.transform.localPosition.x,
314 z = featureObject.transform.localPosition.z
315 }.normalized;
316
317 if (direction.magnitude == 0) return;
318
319 featureObject.transform.forward = direction;
320 }
321
322 if (placement.mirroring)
323 {
324 if (MirroredObj == null)
325 MirroredObj = GameObject.Instantiate(featureObject, featureObject.transform.parent);
326
327 MirroredObj.transform.localScale = MirrorX(featureObject.transform.localScale);
328 MirroredObj.transform.localPosition = MirrorX(featureObject.transform.localPosition);
329 MirroredObj.transform.forward = MirrorX(featureObject.transform.forward);
330 }
331 }

334 {
335 return new Vector3
336 (
337 -vector.x, vector.y, vector.z
338 );
339 }
