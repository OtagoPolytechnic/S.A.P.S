251 public void SetPositionFromPlacement()
252 {
253 if (model == null)
254 {
255 Debug.LogWarning($"Tried setting position of {featureObject.name} but couldn't find the body.");
256 return;
257 }
258
259 if (featureObject == null)
260 {
261 return;
262 }
263
264 if (placement.fixedPosition)
265 {
266 featureObject.transform.localPosition = Vector3.zero;
267 return;
268 }
269
270 BodyMargins margins = model.bodyMargins;
271
272 // get the midpoint of the body, but not above or below the cylindrical area
273 // these two magic numbers are fractionally where capsule caps begin; i.e. where cylinder becomes a semisphere
274 Vector3 originInCylinder = new() { y = Mathf.Clamp(placement.height * margins.height, margins.cylinderBottom, margins.cylinderTop) };
275
276 // account for rounding of the top and bottom of capsule
277 Vector2 distanceFromOrigin = new()
278 {
279 y = Mathf.Abs(placement.height * margins.height - originInCylinder.y)
280 };
281 float radiusScale = 1;
282 if (placement.protruding && distanceFromOrigin.y > 0)
283 {
284 // scales horizontal coords to stick feature to body when height is above or below the capsule crossover point
285 // using trig: hypotenuse = margins.radius, side B = originDistance, looking for length of side A
286 distanceFromOrigin.x = Mathf.Sqrt(Mathf.Pow(margins.radius, 2) - Mathf.Pow(distanceFromOrigin.y, 2));
287 distanceFromOrigin.x = Mathf.Clamp(Mathf.Abs(distanceFromOrigin.x), 0, margins.radius);
288 // accounts for the radius of the horizontal cross section getting smaller as you approach the ends of the capsule
289 radiusScale = Mathf.InverseLerp(0, margins.radius, distanceFromOrigin.x);
290 }
291
292 // roughly get the point on the surface of the body
293 featureObject.transform.localPosition = new(
294 margins.radius * radiusScale * Mathf.Sin(placement.angle),
295 margins.height * placement.height,
296 margins.radius * radiusScale * Mathf.Cos(placement.angle)
297 );
298
299 if (placement.protruding)
300 {
301 // stand up along the angle from origin
302 Vector3 positionOnMesh = featureObject.transform.localPosition;
303 positionOnMesh.y = placement.height * margins.height;
304 featureObject.transform.up = (positionOnMesh - originInCylinder).normalized;
305 }
306 else
307 {
308 // face horizontally away from the midpoint (0, 0)
309 Vector3 direction = new Vector3()
310 {
311 x = featureObject.transform.localPosition.x,
312 z = featureObject.transform.localPosition.z
313 }.normalized;
314
315 if (direction.magnitude == 0) return;
316
317 featureObject.transform.forward = direction;
318 }
319
320 if (placement.mirroring)
321 {
322 if (MirroredObj == null)
323 MirroredObj = GameObject.Instantiate(featureObject, featureObject.transform.parent);
324
325 MirroredObj.transform.localScale = MirrorX(featureObject.transform.localScale);
326 MirroredObj.transform.localPosition = MirrorX(featureObject.transform.localPosition);
327 MirroredObj.transform.forward = MirrorX(featureObject.transform.forward);
328 }
329 }

333 return new Vector3

336 );
