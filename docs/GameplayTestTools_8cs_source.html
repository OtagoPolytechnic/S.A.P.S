201 void ApplyNPCSettings()
202 {
203 if (npcSpawner == null) npcSpawner = FindFirstObjectByType<NPCSpawner>();
204 if (npcSpawner == null) return;
205
206 npcSpawner.crowdPoints.ForEach(p => p.GetComponent<CrowdPointAllocator>().points.ForEach(p2 => p2.GetComponent<MeshRenderer>().enabled = displayCrowdPoints));
207
208 foreach (GameObject p in npcSpawner.crowdPoints)
209 {
210 MeshRenderer mr = p.GetComponent<MeshRenderer>();
211 if (mr != null)
212 {
213 mr.enabled = displayCrowdPoints;
214 }
215 }
216
217 if (!Application.isPlaying) return;
218
219 if (addSuspicion || removeSuspicion || setSuspicion)
220 {
221 foreach (VisionBehaviour vb in FindObjectsByType<VisionBehaviour>(FindObjectsSortMode.None))
222 {
223 vb.Suspicion = setSuspicion ? setSuspicionAmount :
224 addSuspicion ? vb.Suspicion + addSuspicionAmount : vb.Suspicion - addSuspicionAmount;
225 }
226 }
227
228 foreach (NavMeshAgent agent in FindObjectsByType<NavMeshAgent>(FindObjectsSortMode.None))
229 {
230 agent.enabled = !freezeNavMeshAgents;
231 }
232 }

282 async Awaitable TeleportTargetToPlayer()
283 {
284 if (player == null) player = FindPlayer();
285 // disable NavMeshAgent for 1 frame because teleporting freaks it out sometimes
286 targetNPC.GetComponent<NavMeshAgent>().enabled = false;
287 targetNPC.transform.position = player.transform.position + player.transform.forward * teleportTargetPlayerDistance
288 + Vector3.up;
289 Debug.Log($"Teleported {targetNPC.name} to {player.name}");
290 await Awaitable.NextFrameAsync();
291 targetNPC.GetComponent<NavMeshAgent>().enabled = true;
292 }
