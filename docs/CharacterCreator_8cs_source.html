199 void AddAccessories(CharacterModel model)
200 {
201 // make sure not to try add more accessories than the feature pack provides
202 int maxAccessories = Mathf.Clamp(featurePack.maxAccessories, 0, featurePack.accessories.Length + 1);
203 int numAccessories = Random.Range(0, maxAccessories + 1);
204 if (numAccessories == 0)
205 {
206 return;
207 }
208 int[] accessoryIndexes = GetRandomAccessoryIndexes(numAccessories);
209 for (int i = 0; i < accessoryIndexes.Length; i++)
210 {
211 model.AddFeature(featurePack.accessories[accessoryIndexes[i]], new(){fixedPosition = true});
212 }
213 }

222 {
223 int index;
