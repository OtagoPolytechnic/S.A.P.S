170 void AddAccessories(CharacterModel model)
171 {
172 // make sure not to try add more accessories than the feature pack provides
173 int maxAccessories = Mathf.Clamp(featurePack.maxAccessories, 0, featurePack.accessories.Length + 1);
174 int numAccessories = Random.Range(0, maxAccessories + 1);

179 int[] accessoryIndexes = GetRandomAccessoryIndexes(numAccessories);

182 model.AddFeature(featurePack.accessories[accessoryIndexes[i]], new(){fixedPosition = true});
183 }
